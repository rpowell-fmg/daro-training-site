Option 1: Merge then do case_when()

merged_planes<-flights %>% 
  select(tailnum, carrier) %>% 
  left_join(airlines, join_by(carrier)) %>% 
  left_join(planes, join_by(tailnum))


merged_planes %>% 
  select(tailnum, carrier, name, manufacturer) %>% 
  mutate(Airbus = case_when(manufacturer == "AIRBUS INDUSTRIE" ~ 1,

  1. Calculate the average temperature, wind speed, and percipitation by origin airport for the month of January. Try to do this using pivot_longer() and one summarize() statement.
weather %>% 
  filter(month == 1) %>% 
  select(origin, temp, wind_speed, precip) %>% 
  pivot_longer(cols=temp:precip, 
               names_to = "weather_type",
